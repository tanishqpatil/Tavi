cmake_minimum_required(VERSION 3.22)
project(tavi LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(TAVI_OPENCL "Build an optional, explicitly loaded OpenCL module" OFF)
set(BUILD_SHARED_LIBS ${TAVI_OPENCL} CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
option(TAVI_KLEIDIAI "Compile portable, runtime-dispatched CPU kernels" OFF)
set(GGML_CPU_KLEIDIAI ${TAVI_KLEIDIAI} CACHE BOOL "" FORCE)
if (TAVI_KLEIDIAI)
    include(FetchContent)
    set(TAVI_KLEIDIAI_URL https://github.com/ARM-software/kleidiai/releases/download/v1.32.0/kleidiai-v1.32.0-src.tar.gz)
    if (DEFINED ENV{TAVI_KLEIDIAI_ARCHIVE})
        set(TAVI_KLEIDIAI_URL "$ENV{TAVI_KLEIDIAI_ARCHIVE}")
    endif()
    # First declaration wins over the upstream MD5-only declaration.
    FetchContent_Declare(kleidiai
        URL "${TAVI_KLEIDIAI_URL}"
        URL_HASH SHA256=a48d373e7c5d721a0aecbf5ddaf0bdf245bc8064e6b84086f13db0e102b553a6
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
endif()
set(GGML_BACKEND_DL ${TAVI_OPENCL} CACHE BOOL "" FORCE)
set(GGML_OPENCL ${TAVI_OPENCL} CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
if (TAVI_OPENCL)
    set(OPENCL_ICD_LOADER_BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    set(OPENCL_ICD_LOADER_BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(ENABLE_OPENCL_LAYERS OFF CACHE BOOL "" FORCE)
    set(ENABLE_OPENCL_LAYERINFO OFF CACHE BOOL "" FORCE)
    set(OPENCL_ICD_LOADER_HEADERS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/opencl-headers" CACHE PATH "" FORCE)
    add_subdirectory(../../../../native/opencl-loader opencl-link-sdk EXCLUDE_FROM_ALL)
    # Upstream's raw LINK_FLAGS splits source paths containing spaces.
    set_target_properties(OpenCL PROPERTIES VERSION "" SOVERSION "" LINK_FLAGS "")
    target_link_options(OpenCL PRIVATE "LINKER:--version-script,${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/opencl-loader/loader/linux/icd_exports.map")
    set(OpenCL_INCLUDE_DIR "${OPENCL_ICD_LOADER_HEADERS_DIR}" CACHE PATH "" FORCE)
    set(OpenCL_LIBRARY OpenCL CACHE STRING "" FORCE)
endif()
add_subdirectory(../../../../native/llama.cpp llama-build)
add_library(tavi SHARED tavi.cpp)
target_link_libraries(tavi PRIVATE llama mtmd android log)
target_link_options(tavi PRIVATE "-Wl,-z,max-page-size=16384")
if (TAVI_OPENCL)
    target_compile_definitions(tavi PRIVATE TAVI_DYNAMIC_BACKENDS=1)
    add_dependencies(tavi ggml-cpu ggml-opencl)
    foreach(lib ggml ggml-base ggml-cpu ggml-opencl llama mtmd)
        set_target_properties(${lib} PROPERTIES VERSION "" SOVERSION "")
        target_link_options(${lib} PRIVATE "-Wl,-z,max-page-size=16384")
    endforeach()
endif()
